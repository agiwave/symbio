### 1.2 核心设计理念

| 特性 | 说明 |
|------|------|
| **三命令接口** | `meta` / `invoke` / `stream` — 所有插件能力统一通过这三个命令访问 |
| **Connect 机制** | `connect` / `connect.send` / `connect.close` — 双向持久连接，支持实时推送 |
| **路径寻址** | 空路径访问根插件，`["plugin_name"]` 访问指定插件，支持分形嵌套 |
| **插件平等** | 所有插件实现统一的 `Plugin` 接口，Root 也是 `Arc<dyn Plugin>` |
| **依赖注入** | `EventSender` trait 实现平台无关事件机制，支持 Tauri/Web/CLI 等 |

### 1.3 插件接口

pub struct CognitiveUnit {
    pub id: String,
    #[serde(rename = "type")]
    pub unit_type: CognitiveUnitType,
    pub content: Content,
    #[serde(default)]
    pub context: Context,
    #[serde(default)]
    pub strength: Strength,
    pub source: Source,
    #[serde(default)]
    pub usage: Usage,
    #[serde(default)]
    pub generality: Generality,
    #[serde(default)]
    pub relations: Vec<Relation>,
}

impl CognitiveUnit {
    /// 创建新的认知单元
    pub fn new(
        unit_type: CognitiveUnitType,
        summary: String,
        body: serde_yaml::Value,
    ) -> Self {
        let now = Utc::now();
        let type_prefix = unit_type.to_string();
        let id = format!("{}_{}", type_prefix, now.format("%Y%m%d_%H%M%S"));
        
        Self {
            id,
            unit_type,
            content: Content {
                summary,
                body,
                format: ContentFormat::Structured,
            },
            context: Context::default(),
            strength: Strength::default(),
            source: Source {
                origin: SourceOrigin::Imported,
                evidence: vec![],
                created_at: now,
                updated_at: now,
            },
            usage: Usage::default(),
            generality: Generality::default(),
            relations: vec![],
        }
    }
    
    /// 计算检索得分
    pub fn retrieval_score(&self, semantic_similarity: f32, activation: f32) -> f32 {
        let importance_factor = self.strength.importance as f32 / 5.0;
        let accessibility_factor = self.strength.accessibility;
        
        0.4 * semantic_similarity 
            + 0.3 * activation 
            + 0.2 * importance_factor 
            + 0.1 * accessibility_factor
    }
    
    /// 更新访问记录
    pub fn record_access(&mut self) {
        self.usage.access_count += 1;
        self.usage.last_accessed = Some(Utc::now());
        
        // 提升可访问性
        self.strength.accessibility = (self.strength.accessibility + 0.1).min(1.0);
    }
    
    /// 应用时间衰减
    pub fn apply_decay(&mut self, decay_rate: f32) {
        self.strength.accessibility *= (1.0 - decay_rate);
        
        // 最小可访问性
        if self.strength.accessibility < 0.1 {
            self.strength.accessibility = 0.1;
        }
    }
}

use serde::{Deserialize, Serialize};

/// 向量嵌入
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Embedding {
    pub unit_id: String,
    pub vector: Vec<f32>,
    pub model: String,
    pub created_at: chrono::DateTime<chrono::Utc>,
}

impl Embedding {
    pub fn new(unit_id: String, vector: Vec<f32>, model: String) -> Self {
        Self {
            unit_id,
            vector,
            model,
            created_at: chrono::Utc::now(),
        }
    }
    
    /// 计算余弦相似度
    pub fn cosine_similarity(&self, other: &[f32]) -> f32 {
        let dot: f32 = self.vector.iter()
            .zip(other.iter())
            .map(|(a, b)| a * b)
            .sum();
        
        let norm_a: f32 = self.vector.iter().map(|x| x * x).sum::<f32>().sqrt();
        let norm_b: f32 = other.iter().map(|x| x * x).sum::<f32>().sqrt();
        
        if norm_a == 0.0 || norm_b == 0.0 {
            0.0
        } else {
            dot / (norm_a * norm_b)
        }
    }
}

### 3.1 认知单元存储

```rust
// src/storage/unit_store.rs

use anyhow::{Context, Result};
use std::path::{Path, PathBuf};
use std::collections::HashMap;
use tokio::fs;
use tokio::io::AsyncWriteExt;

use crate::models::CognitiveUnit;

/// 认知单元存储
pub struct UnitStore {
    base_path: PathBuf,
    units: HashMap<String, CognitiveUnit>,
    by_type: HashMap<String, Vec<String>>, // 类型索引
}

impl UnitStore {
    pub async fn new(base_path: impl AsRef<Path>) -> Result<Self> {
        let base_path = base_path.as_ref().to_path_buf();
        let mut store = Self {
            base_path,
            units: HashMap::new(),
            by_type: HashMap::new(),
        };
        
        store.load_all().await?;
        Ok(store)
    }
    
    /// 加载所有认知单元
    async fn load_all(&mut self) -> Result<()> {
        let units_path = self.base_path.join("units");
        
        if !units_path.exists() {
            fs::create_dir_all(&units_path).await?;
            return Ok(());
        }
        
        let mut entries = fs::read_dir(&units_path).await?;
        
        while let Some(entry) = entries.next_entry().await? {
            let path = entry.path();
            if path.extension().map(|e| e == "yaml").unwrap_or(false) {
                if let Ok(unit) = self.load_unit(&path).await {
                    let type_key = unit.unit_type.to_string();
                    self.by_type.entry(type_key).or_default().push(unit.id.clone());
                    self.units.insert(unit.id.clone(), unit);
                }
            }
        }
        
        tracing::info!("Loaded {} cognitive units", self.units.len());
        Ok(())
    }
    
    /// 加载单个认知单元
    async fn load_unit(&self, path: &Path) -> Result<CognitiveUnit> {
        let content = fs::read_to_string(path).await?;
        let unit: CognitiveUnit = serde_yaml::from_str(&content)
            .with_context(|| format!("Failed to parse unit from {:?}", path))?;
        Ok(unit)
    }
    
    /// 保存认知单元
    pub async fn save(&self, unit: &CognitiveUnit) -> Result<()> {
        let units_path = self.base_path.join("units");
        fs::create_dir_all(&units_path).await?;
        
        let file_path = units_path.join(format!("{}.yaml", unit.id));
        let content = serde_yaml::to_string(unit)?;
        
        let mut file = fs::File::create(&file_path).await?;
        file.write_all(content.as_bytes()).await?;
        
        tracing::debug!("Saved unit {} to {:?}", unit.id, file_path);
        Ok(())
    }
    
    /// 获取认知单元
    pub fn get(&self, id: &str) -> Option<&CognitiveUnit> {
        self.units.get(id)
    }
    
    /// 获取所有ID
    pub fn all_ids(&self) -> impl Iterator<Item = &String> {
        self.units.keys()
    }
    
    /// 按类型获取
    pub fn get_by_type(&self, type_name: &str) -> Vec<&CognitiveUnit> {
        self.by_type
            .get(type_name)
            .map(|ids| ids.iter().filter_map(|id| self.units.get(id)).collect())
            .unwrap_or_default()
    }
    
    /// 添加或更新认知单元
    pub async fn upsert(&mut self, unit: CognitiveUnit) -> Result<()> {
        let id = unit.id.clone();
        let type_key = unit.unit_type.to_string();
        
        // 更新类型索引
        if !self.units.contains_key(&id) {
            self.by_type.entry(type_key).or_default().push(id.clone());
        }
        
        // 保存到文件
        self.save(&unit).await?;
        
        // 更新内存
        self.units.insert(id, unit);
        
        Ok(())
    }
}
```

// src/storage/vector_store.rs

use anyhow::Result;
use qdrant_client::prelude::*;
use qdrant_client::qdrant::{SearchPoints, PointStruct, Value};

use crate::models::Embedding;

/// 向量存储配置
pub struct VectorStoreConfig {
    pub url: String,
    pub collection_name: String,
    pub vector_size: usize,
}

impl Default for VectorStoreConfig {
    fn default() -> Self {
        Self {
            url: "http://localhost:6333".to_string(),
            collection_name: "cognitive_units".to_string(),
            vector_size: 1536, // OpenAI text-embedding-3-small
        }
    }
}

/// 向量存储
pub struct VectorStore {
    client: Qdrant,
    config: VectorStoreConfig,
}

impl VectorStore {
    pub async fn new(config: VectorStoreConfig) -> Result<Self> {
        let client = Qdrant::from_url(&config.url).build()?;
        
        // 确保集合存在
        let collections = client.list_collections().await?;
        let exists = collections.collections.iter()
            .any(|c| c.name == config.collection_name);
        
        if !exists {
            client.create_collection(&CreateCollection {
                collection_name: config.collection_name.clone(),
                vectors_config: Some(VectorsConfig {
                    config: Some(vectors_config::Config::Params(VectorParams {
                        size: config.vector_size as u64,
                        distance: Distance::Cosine.into(),
                        ..Default::default()
                    })),
                }),
                ..Default::default()
            }).await?;
            
            tracing::info!("Created collection: {}", config.collection_name);
        }
        
        Ok(Self { client, config })
    }
    
    /// 存储向量
    pub async fn store(&self, embedding: &Embedding) -> Result<()> {
        let point = PointStruct {
            id: Some(embedding.unit_id.clone().into()),
            vectors: Some(embedding.vector.clone().into()),
            payload: HashMap::from([
                ("model".to_string(), Value::from(embedding.model.clone())),
                ("created_at".to_string(), Value::from(embedding.created_at.to_rfc3339())),
            ]),
        };
        
        self.client.upsert_points(
            UpsertPoints {

use std::collections::{HashMap, BinaryHeap};
use std::cmp::Ordering;

use crate::models::{CognitiveUnit, RelationType};
use crate::storage::UnitStore;

/// 激活节点
#[derive(Debug, Clone)]
struct ActivationNode {
    unit_id: String,
    activation: f32,
}

impl PartialEq for ActivationNode {
    fn eq(&self, other: &Self) -> bool {
        self.activation == other.activation
    }
}

impl Eq for ActivationNode {}

impl PartialOrd for ActivationNode {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        Some(self.cmp(other))
    }
}

impl Ord for ActivationNode {
    fn cmp(&self, other: &Self) -> Ordering {
        // 最大堆：激活值大的优先
        other.activation.partial_cmp(&self.activation).unwrap_or(Ordering::Equal)
    }
}

/// 检索配置 (支持元级热更新/Self-Modifying)
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RetrievalConfig {
    /// 初始激活值
    pub initial_activation: f32,
    /// 每跳衰减因子
    pub decay_factor: f32,
    /// 最大扩散跳数
    pub max_hops: usize,
    /// 激活阈值
    pub activation_threshold: f32,
    /// 最大返回数量
    pub max_results: usize,
    
    // ======== AGI 终极破极化机制新增参数 ========
    /// 允许自我重写标志（元级学习）
    pub is_mutable: bool,
    /// 心智社会沙盘推演硬件花销限制
    pub cost_awareness_limit: u32,
}

impl Default for RetrievalConfig {
    fn default() -> Self {
        Self {
            initial_activation: 1.0,
            decay_factor: 0.7,
            max_hops: 3,
            activation_threshold: 0.1,
            max_results: 20,
            is_mutable: true,            // 允许底层参数随经验反思自我进化
            cost_awareness_limit: 1000,  // 防止心智辩论消耗过度算力
        }
    }
}

/// 检索引擎
pub struct RetrievalEngine {
    unit_store: UnitStore,
    config: RetrievalConfig,
}

impl RetrievalEngine {
    pub fn new(unit_store: UnitStore, config: RetrievalConfig) -> Self {
        Self { unit_store, config }
    }
    
    /// 执行检索
    pub async fn retrieve(&self, query: &str) -> Vec<(String, f32)> {
        // 1. 向量检索获取初始候选
        let candidates = self.vector_search(query).await;
        
        // 2. 激活扩散
        let activations = self.spreading_activation(candidates);
        
        // 3. 排序并返回
        let mut results: Vec<_> = activations.into_iter()
            .filter(|(_, activation)| *activation > self.config.activation_threshold)
            .collect();
        
        results.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(Ordering::Equal));
        results.truncate(self.config.max_results);
        
        results
    }
    
    /// 向量检索（简化实现）
    async fn vector_search(&self, _query: &str) -> Vec<(String, f32)> {
        // 实际实现需要调用向量数据库
        // 这里返回空，实际使用时需要注入向量存储
        vec![]
    }
    
    /// 激活扩散算法
    fn spreading_activation(&self, initial: Vec<(String, f32)>) -> HashMap<String, f32> {
        let mut activations: HashMap<String, f32> = HashMap::new();
        let mut heap: BinaryHeap<ActivationNode> = BinaryHeap::new();
        
        // 初始化
        for (id, sim) in initial {
            let activation = sim * self.config.initial_activation;
            activations.insert(id.clone(), activation);
            heap.push(ActivationNode { unit_id: id, activation });
        }
        
        // 扩散
        let mut visited_hops: HashMap<String, usize> = HashMap::new();
        
        while let Some(node) = heap.pop() {
            let hops = *visited_hops.get(&node.unit_id).unwrap_or(&0);
            
            if hops >= self.config.max_hops {
                continue;
            }
            
            if let Some(unit) = self.unit_store.get(&node.unit_id) {
                for relation in &unit.relations {
                    let relation_strength = match relation.relation_type {
                        RelationType::CounterExample | RelationType::Contradicts => {
                            // 负面关系降低激活
                            -relation.strength * 0.5
                        }
                        _ => relation.strength,
                    };
                    
                    let new_activation = node.activation 
                        * self.config.decay_factor 
                        * relation_strength
                        * relation.weight;
                    
                    if new_activation > self.config.activation_threshold {
                        let current = activations.get(&relation.target_id).copied().unwrap_or(0.0);
                        let updated = current + new_activation;
                        
                        activations.insert(relation.target_id.clone(), updated);
                        
                        let target_hops = *visited_hops.get(&relation.target_id).unwrap_or(&usize::MAX);
                        if hops + 1 < target_hops {
                            visited_hops.insert(relation.target_id.clone(), hops + 1);
                            heap.push(ActivationNode {
                                unit_id: relation.target_id.clone(),
                                activation: updated,
                            });
                        }
                    }
                }
            }
        }
        
        activations
    }
}

        for unit in units {

            usage_frequency_weight: 0.05,
            time_decay_rate: 0.001,

    config: LearningConfig,
}

impl LearningEngine {
    pub fn new(config: LearningConfig) -> Self {

    /// 应用时间衰减
    pub fn apply_time_decay(&self, unit: &mut CognitiveUnit) {
        let now = Utc::now();
        
        if let Some(last_accessed) = unit.usage.last_accessed {
            let days_since_access = (now - last_accessed).num_days() as f32;
            let total_decay = self.config.time_decay_rate * days_since_access;
            
            unit.strength.accessibility *= (1.0 - total_decay).max(0.1);
        }
    }
    
    /// 处理正面反馈
    pub fn apply_positive_feedback(&self, unit: &mut CognitiveUnit) {
        unit.usage.feedback_positive += 1;
        
        let feedback_boost = self.config.positive_feedback_weight;
        let usage_boost = if unit.usage.access_count > 10 {
            self.config.usage_frequency_weight * 0.5
        } else {
            0.0
        };
        
        unit.strength.importance = ((unit.strength.importance as f32 + feedback_boost + usage_boost)
            .min(5.0)) as u8;
        unit.strength.confidence = (unit.strength.confidence + feedback_boost).min(1.0);
    }
    
    /// 处理负面反馈
    pub fn apply_negative_feedback(&self, unit: &mut CognitiveUnit) {
        unit.usage.feedback_negative += 1;
        
        let feedback_penalty = self.config.negative_feedback_weight;
        
        unit.strength.confidence = (unit.strength.confidence - feedback_penalty).max(0.1);
        
        // 如果负面反馈过多，降低重要性
        if unit.usage.feedback_negative > unit.usage.feedback_positive {
            unit.strength.importance = (unit.strength.importance.saturating_sub(1)).max(1);
        }
    }
    
    /// 记录使用
    pub fn record_usage(&self, unit: &mut CognitiveUnit) {
        unit.record_access();
        
        // 使用频率提升
        if unit.usage.access_count % 10 == 0 {
            unit.strength.importance = (unit.strength.importance + 1).min(5);
        }
    }

#[derive(Deserialize)]
struct SearchRequest {
    query: String,
    limit: Option<usize>,
}

#[derive(Serialize)]
struct SearchResult {
    unit_id: String,
    score: f32,
    summary: String,
}

async fn search_units(
    State(state): State<AppState>,
    Json(req): Json<SearchRequest>,
) -> Json<Vec<SearchResult>> {
    let limit = req.limit.unwrap_or(10);
    let results = state.retrieval_engine.retrieve(&req.query).await;
    
    let search_results: Vec<_> = results.into_iter()
        .filter_map(|(id, score)| {
            state.unit_store.get(&id).map(|unit| SearchResult {
                unit_id: id,
                score,
                summary: unit.content.summary.clone(),
            })
        })
        .take(limit)
        .collect();
    
    Json(search_results)
}

// ========== 对话 ==========

#[derive(Deserialize)]
struct ChatRequest {
    message: String,
    context: Option<Vec<String>>,
}

#[derive(Serialize)]
struct ChatResponse {
    reply: String,
    sources: Vec<String>,
}

async fn chat(
    State(state): State<AppState>,
    Json(req): Json<ChatRequest>,
) -> Result<Json<ChatResponse>, StatusCode> {
    // 1. 检索相关认知单元
    let results = state.retrieval_engine.retrieve(&req.message).await;
    
    // 2. 获取认知单元
    let units: Vec<_> = results.iter()
        .filter_map(|(id, _)| state.unit_store.get(id).cloned())
        .collect();
    
    // 3. 构建上下文
    let context = state.injection_engine.build_context(&units);
    
    // 4. 构建消息
    let messages = vec![
        Message {
            role: "system".to_string(),
            content: format!("你是一位专家助手。{}\n\n{}", context, "请基于以上专家知识回答用户问题。"),
        },
        Message {
            role: "user".to_string(),
            content: req.message,
        },
    ];
    
    // 5. 调用LLM
    let response = state.llm_adapter.chat(messages).await
        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;
    
    // 6. 返回结果
    Ok(Json(ChatResponse {
        reply: response.content,
        sources: results.into_iter().take(5).map(|(id, _)| id).collect(),
    }))
}
